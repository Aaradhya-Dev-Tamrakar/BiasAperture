\begin{verbatim}
# Run complete audit across demographic strata:
uv run bias-aperture audit \
  -i data/processed/fairface_predictions_val.csv \
  -a race \
  --true-label-col true_gender \
  --predicted-label-col predicted_gender \
  --race-col subgroup_race \
  --backend dual \
  --bca-resamples 1000 \
  -o report/audit_val_race_verified.html
\end{verbatim}
      \end{card}

      \vspace{0.1cm}
      \begin{statcard}{Execution Pipeline Stages}
        \scriptsize
        \begin{enumerate}\scriptsize
          \item \textbf{Schema Validation:} Ingests and verifies 10,954 rows.
          \item \textbf{Dual Compute:} Runs Fairlearn + AIF360 strategies.
          \item \textbf{Inference Engine:} Computes $\chi^2$ and $B=1,000$ BCa CIs.
          \item \textbf{Explainability:} Evaluates surrogate attribution on disparities.
          \item \textbf{Report Assembly:} Inlines SVG charts into offline HTML.
        \end{enumerate}
      \end{statcard}
